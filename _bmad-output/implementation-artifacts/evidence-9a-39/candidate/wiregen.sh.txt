#!/bin/bash
# schemas/README.md regeneration procedure, writing only into the scratch directory $G.
# Usage: wiregen.sh <repo root> <flatc> <scratch dir>
set -e
ROOT="$1"; FLATC="$2"; G="$3"
mkdir -p "$G/base-schema" "$G/base-cs" "$G/base-ts" "$G/new-cs" "$G/new-ts"
git -C "$ROOT" show HEAD:schemas/workshop_wire.fbs > "$G/base-schema/workshop_wire.fbs"
gen_cs() {
  "$FLATC" --csharp -o "$2" "$1"
  {
  printf '%s\n' '// <auto-generated>' '//  automatically generated by the FlatBuffers compiler, do not modify' '// </auto-generated>' '' \
    'namespace CuriousContraptions.Wire' '{' 'using global::System;' 'using global::System.Collections.Generic;' \
    'using global::Google.FlatBuffers;' ''
  first=1
  for f in $(ls "$2/CuriousContraptions/Wire" | sort); do
    [ $first -eq 0 ] && echo; first=0
    awk 'started && /^}$/ { last = NR } { lines[NR] = $0 } /^\{$/ && !started { start = NR + 1; started = 1 }
         END { while (lines[start] == "" || lines[start] ~ /^using /) start++;
               end = last - 1; while (end >= start && lines[end] == "") end--; for (i = start; i <= end; i++) print lines[i] }' \
      "$2/CuriousContraptions/Wire/$f"
  done
  echo; echo '}'
  } > "$3"
}
gen_ts() {
  "$FLATC" --ts -o "$2" "$1"
  {
  echo '// automatically generated by the FlatBuffers compiler, do not modify'
  echo 'import * as flatbuffers from "flatbuffers";'
  echo
  for name in $(grep -E '^(enum|struct|table) ' "$1" | awk '{print $2}'); do
    file=$(echo "$name" | sed -E 's/([a-z0-9])([A-Z])/\1-\2/g' | tr 'A-Z' 'a-z')
    grep -v -E '^// automatically generated|^import ' "$2/curious-contraptions/wire/$file.ts" | cat -s | sed '1{/^$/d;}'
    echo
  done
  } > "$3"
}
gen_cs "$G/base-schema/workshop_wire.fbs" "$G/base-cs" "$G/base.cs"
gen_ts "$G/base-schema/workshop_wire.fbs" "$G/base-ts" "$G/base.ts"
gen_cs "$ROOT/schemas/workshop_wire.fbs" "$G/new-cs" "$G/new.cs"
gen_ts "$ROOT/schemas/workshop_wire.fbs" "$G/new-ts" "$G/new.ts"
echo "flatc: $("$FLATC" --version) sha256 $(shasum -a 256 "$FLATC" | cut -d' ' -f1)"
cmp "$G/base.cs" <(git -C "$ROOT" show HEAD:engine/gpu/WorkshopWireGenerated.cs) && echo "CS base reproduction: IDENTICAL"
MARK=$(git -C "$ROOT" show HEAD:tools/e2e/workshop-wire.ts | grep -n "hand-written helpers" | head -1 | cut -d: -f1)
echo "TS helpers marker at line $MARK"
