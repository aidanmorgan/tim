#pragma warning disable CA1859
using System.Security.Cryptography;
using CuriousContraptions.Gpu;

namespace CuriousContraptions.Tests;

/// <summary>Reviewer probe (throwaway, C-Bumper+Spring): independent lesson-hash sample, identical on the record path (3fce7a03) and the
/// candidate except the two placement lines marked ADAPTER. Hashes admission bytes, every body read over 1200 ticks, every contact-work
/// read, the Reset state and the goal verdict; prints the final contact-work reads so a paid route is visible.</summary>
public sealed class ReviewLessonHashProbe
{
    private static readonly System.Text.Json.JsonSerializerOptions Options = new() { IncludeFields = true };
    private static string Hash(string text) => Convert.ToHexString(SHA256.HashData(System.Text.Encoding.UTF8.GetBytes(text)));

    private static WorkshopConstruction Lesson(string id, double precision) => PuzzleCatalogue.Create(new PuzzleId(id), new(2),
        WorkshopCadenceSettings.Default(), new(1), new((Half)precision));

    // ADAPTER (candidate generic path)
    private static IWorkshopInstance Bumper(GpuBodyId id, double x, double y, double z, float s) =>
        WorkshopInput.Generic(PartId.PinballBumper, id, x, y, z, 0, 0, 0, 1, new([new(new("strength"), s)]));
    private static IWorkshopInstance Spring(GpuBodyId id, double x, double y, double z, double qx, double qy, double qz, double qw,
        float q, float k = 400f, float c = .2f) => WorkshopInput.Generic(PartId.Springboard, id, x, y, z, qx, qy, qz, qw,
        new([new(new("stiffness"), k), new(new("damping"), c), new(new("initial_compression"), q)]));

    [Fact]
    public async Task WriteSample()
    {
        const double Qz = -.11166252195835114, Qw = .9937461614608765;
        var routes = new (string Name, WorkshopConstruction Construction)[]
        {
            ("spring_forward charged 1", Lesson("spring_forward", 1).WithInstance(Spring(new(3), -3.315333366394043, 3.1000001430511475, 0, 0, 0, Qz, Qw, .2f))),
            ("spring_forward charged 0.45", Lesson("spring_forward", .45).WithInstance(Spring(new(3), -3.315333366394043, 3.1000001430511475, 0, 0, 0, Qz, Qw, .2f))),
            ("spring_forward custom k=900 c=1.5 q=0.13 1", Lesson("spring_forward", 1).WithInstance(Spring(new(3), -3.315333366394043, 3.1000001430511475, 0, 0, 0, Qz, Qw, .13f, 900f, 1.5f))),
            ("bumper_sidekick authored-row 3.3 s8 1", Lesson("bumper_sidekick", 1).WithInstance(Bumper(new(3), -3.3, 1.5, 0, 8f))),
            ("bumper_sidekick authored-row 3.3 s8 0.45", Lesson("bumper_sidekick", .45).WithInstance(Bumper(new(3), -3.3, 1.5, 0, 8f))),
            ("bumper_sidekick 3.3 s20 0.45", Lesson("bumper_sidekick", .45).WithInstance(Bumper(new(3), -3.3, 1, 0, 20f))),
            ("bumper_sidekick 3.3 s3.3333333 0", Lesson("bumper_sidekick", 0).WithInstance(Bumper(new(3), -3.3, 1, 0, 3.3333333f))),
            ("bumper_depth authored-row 3.3 s8 0.45", Lesson("bumper_depth", .45).WithInstance(Bumper(new(3), 0, 1.5, 3.3, 8f))),
            ("ready_to_rebound receiver 0.45", Lesson("ready_to_rebound", .45).WithInstance(WorkshopInput.Generic(PartId.Receiver, new(3), 1.8, 1.3, 0, 0, 0, 0, 1))),
        };
        var lines = new List<string>();
        foreach (var (name, construction) in routes)
        {
            var scene = WorkshopPhysicsCompiler.Compile(construction, new(1, 2));
            var bytes = PhysicsGpuAbi.Admission(scene, new(1), new(SimulationCadence.Hz120, PhysicalStepProfile.Canonical480Hz, new(1)));
            var (_, reads, reset) = await NoFixedCapacityTests.RunAndReset(construction, 1200);
            var trace = System.Text.Json.JsonSerializer.Serialize(reads.Select(read => Enumerable.Range(0, read.Bodies.Count)
                .Select(i => read.Bodies[i]).ToArray()).ToArray(), Options);
            var works = reads.Select(read => Enumerable.Range(0, read.ContactWorks.Count).Select(i => read.ContactWorks[i]).ToArray()).ToArray();
            var resetState = System.Text.Json.JsonSerializer.Serialize(Enumerable.Range(0, reset.Bodies.Count).Select(i => reset.Bodies[i]).ToArray(), Options);
            var solved = reads.Any(read => WorkshopGoalEvaluator.Occurrence(construction.Puzzle.Goal, read.Captures, read.Activations) is not null);
            lines.Add($"{name}: admission={Convert.ToHexString(SHA256.HashData(bytes))} trace1200={Hash(trace)} " +
                $"works={Hash(System.Text.Json.JsonSerializer.Serialize(works, Options))} reset={Hash(resetState)} solved={solved}");
            lines.Add($"  first works: {System.Text.Json.JsonSerializer.Serialize(works[0], Options)}");
            lines.Add($"  final works: {System.Text.Json.JsonSerializer.Serialize(works[^1], Options)}");
        }
        await File.WriteAllLinesAsync(Environment.GetEnvironmentVariable("PROBE_OUT")!, lines, TestContext.Current.CancellationToken);
    }
}
