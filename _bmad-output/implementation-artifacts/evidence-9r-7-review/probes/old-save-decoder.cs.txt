using System;
using System.IO;
using System.Linq;
using CuriousContraptions.Gpu;

namespace CuriousContraptions.Tests;

public sealed class M14OldSaveProbe
{
    private const string Scratch = "<scratchpad>/m14/";

    [Fact]
    public void OldVersionedSavesAreTheTypedRefusal()
    {
        var results = File.ReadAllLines(Scratch + "old-saves-v13.txt").Select(line =>
        {
            var parts = line.Split(' ');
            try
            {
                var saved = WorkshopSaveCodec.Decode(Convert.FromBase64String(parts[1]));
                return $"{parts[0]} ADMITTED next {saved.NextBodyId.Value}: {saved.Construction}";
            }
            catch (WorkshopEditRejectedException error) { return $"{parts[0]} REFUSED {error.Reason}: {error.Message}"; }
#pragma warning disable CA1031 // review probe: record any other exception type as a finding
            catch (Exception error) { return $"{parts[0]} THREW {error.GetType().Name}: {error.Message}"; }
#pragma warning restore CA1031
        }).ToArray();
        File.WriteAllLines(Scratch + "old-saves-decoded.txt", results);
        Assert.All(results, result => Assert.Contains(" REFUSED UnsupportedSave", result));
    }
}
