using System;
using System.IO;
using CuriousContraptions.Gpu;

namespace CuriousContraptions.Tests;

public sealed class M14OldSaveProbe
{
    [Fact]
    public void WriteOldSaves()
    {
        var settings = WorkshopCadenceSettings.Default();
        var precision = new PuzzlePrecision((Half)0.45);
        var free = new WorkshopConstruction(new(1), settings, new(WorkshopInput.Basketball(new(1), 0, 6, 0, 0, 0, 0, 1)));
        var first = FirstPrinciples.Create(new(1), settings, new(1), new(2), precision);
        var quarter = QuarterBend.Create(new(1), settings, new(1), new(2), precision);
        var lines = new[]
        {
            "free " + Convert.ToBase64String(WorkshopSaveCodec.Encode(new(free, new(2)))),
            "first_principles " + Convert.ToBase64String(WorkshopSaveCodec.Encode(new(first, new(3)))),
            "quarter_bend " + Convert.ToBase64String(WorkshopSaveCodec.Encode(new(quarter, new(3)))),
        };
        File.WriteAllLines("<scratchpad>/m14/old-saves-v13.txt", lines);
        Assert.Equal(13u, WorkshopSaveCodec.Version);
    }
}
