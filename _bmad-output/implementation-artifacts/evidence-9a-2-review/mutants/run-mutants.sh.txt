#!/usr/bin/env bash
# Murdoch-22 mutants for Story 9A.2 (review-only evidence). Runs in the disposable copy .claude/worktrees/murdoch22-mut,
# detached at 0519cedb. Each mutant edits one file, builds the solution (Debug), runs the whole CuriousContraptions.tests
# project (and the node harnesses for script mutants), records the counts, then restores with git reset --hard + git clean.
# Usage: bash run-mutants.sh.txt <out-dir>
set -u
WT=/Users/aidan/dev/personal/tim/.claude/worktrees/murdoch22-mut
OUT=$1
cd "$WT" || exit 2
[ "$(git rev-parse HEAD)" = "0519cedb3c103fa7ed4eb504e3106c2451042984" ] || { echo "wrong head"; exit 2; }

restore() { git reset --hard -q HEAD; git clean -fdq; }

# sub FILE FROM TO: literal replacement of the first occurrence; fails if FROM is absent.
sub() {
  FROM="$2" TO="$3" perl -0pi -e 'BEGIN{$n=0} $n += s/\Q$ENV{FROM}\E/$ENV{TO}/' "$1"
  git diff --quiet -- "$1" && { echo "ANCHOR NOT FOUND in $1"; return 1; }
  return 0
}

run() { # run NAME NODE(0|1)
  local name=$1 node=$2 log="$OUT/$1.txt"
  {
    echo "# mutant $name  head=$(git rev-parse HEAD)  $(date -u +%FT%TZ)"
    git diff --stat; git diff | head -60
    git status --porcelain
    echo "## build"
    dotnet build CuriousContraptions.slnx -nologo -v q 2>&1 | grep -E "error|Warning\(s\)|Error\(s\)" | head -20
    echo "build_exit=${PIPESTATUS[0]}"
    echo "## dotnet test CuriousContraptions.tests"
    dotnet test CuriousContraptions.tests --no-build -nologo 2>&1 | grep -E "^\s+Failed |^(Passed|Failed)!|error" | head -40
    if [ "$node" = 1 ]; then
      echo "## node harnesses"
      node --experimental-vm-modules --test tools/*.test.mjs 2>&1 | grep -E "^not ok|^# (pass|fail)|ℹ (pass|fail)" | head -20
    fi
    echo "# end $(date -u +%FT%TZ)"
  } > "$log" 2>&1
  restore
  echo "$name done: $(grep -E '^(Passed|Failed)!' "$log" | tr '\n' ' ') build=$(grep build_exit "$log")" | tee -a "$OUT/summary.txt"
}

restore
