#!/usr/bin/env bash
# Murdoch-22 second batch (advisor-requested): a swallowing/retrying catch around each schedule-endpoint decode. Same procedure as
# run-mutants.sh.txt (build the solution, run all of CuriousContraptions.tests, restore with git reset --hard + git clean).
set -u
WT=/Users/aidan/dev/personal/tim/.claude/worktrees/murdoch22-mut
OUT=$1
cd "$WT" || exit 2
[ "$(git rev-parse HEAD)" = "0519cedb3c103fa7ed4eb504e3106c2451042984" ] || { echo "wrong head"; exit 2; }
restore() { git reset --hard -q HEAD; git clean -fdq; }
sub() {
  FROM="$2" TO="$3" perl -0pi -e 's/\Q$ENV{FROM}\E/$ENV{TO}/' "$1"
  git diff --quiet -- "$1" && { echo "ANCHOR NOT FOUND in $1"; return 1; }
  return 0
}
run() {
  local name=$1 log="$OUT/$1.txt"
  {
    echo "# mutant $name  head=$(git rev-parse HEAD)  $(date -u +%FT%TZ)"
    git diff --stat; git diff | head -40
    echo "## build"
    dotnet build CuriousContraptions.slnx -nologo -v q 2>&1 | grep -E "error|Warning\(s\)|Error\(s\)" | head -20
    echo "build_exit=${PIPESTATUS[0]}"
    echo "## dotnet test CuriousContraptions.tests"
    dotnet test CuriousContraptions.tests --no-build -nologo 2>&1 | grep -E "^\s+Failed |^(Passed|Failed)!|error" | head -40
    echo "# end $(date -u +%FT%TZ)"
  } > "$log" 2>&1
  restore
  echo "$name done: $(grep -E '^(Passed|Failed)!' "$log" | tr '\n' ' ') build=$(grep build_exit "$log")" | tee -a "$OUT/summary.txt"
}
restore
END='                var endpoint = WorkshopWireFlatBuffers.DecodeResponse(data[WorkshopScheduleWire.PrepareFieldBytes..]);'
sub CuriousContraptions.Animation.Worker/Program.cs "$END" '                WorkshopResponse endpoint;
                try { endpoint = WorkshopWireFlatBuffers.DecodeResponse(data[WorkshopScheduleWire.PrepareFieldBytes..]); }
                catch (WireFaultException) { return; }' && run R14-readd-animation-endpoint-swallow
sub engine/gpu/BrowserWorkshopClient.Schedule.cs "$END" '                WorkshopResponse endpoint;
                try { endpoint = WorkshopWireFlatBuffers.DecodeResponse(data[WorkshopScheduleWire.PrepareFieldBytes..]); }
                catch (WireFaultException) { endpoint = WorkshopWireFlatBuffers.DecodeResponse(data[(WorkshopScheduleWire.PrepareFieldBytes + 4)..]); }' && run R15-readd-client-endpoint-retry
restore
echo "batch 2 final status lines: $(git status --porcelain | wc -l | tr -d ' ')" | tee -a "$OUT/summary.txt"
