#!/bin/bash
# C-Gate review mutation probes, run in the throwaway candidate copy only. Usage: mutate.sh <out-dir>
set -u
R=/private/tmp/claude-501/-Users-aidan-dev-personal-tim/7e56a042-6670-44d4-ae0f-4efd972de00e/scratchpad
OUT="$1"; mkdir -p "$OUT"
cd "$R/wt-cand"
restore() {
  cp "$R/orig/WorkshopPhysicsEngine.cs" "$R/orig/PrismaticConstraintSolver.cs" CuriousContraptions.Simulation/
  cp "$R/orig/PhysicsPresentationRead.cs" "$R/orig/SupplyRoute.cs" "$R/orig/PrismaticConstraint.cs" engine/gpu/
  cp "$R/orig/BundleValidator.cs" CuriousContraptions.Bundle/
}
run() { # name file perl-expr project filter
  local name="$1" file="$2" expr="$3" project="$4" filter="$5"
  restore
  perl -0pi -e "$expr" "$file"
  if git diff --quiet a598a09e -- "$file" && false; then :; fi
  if cmp -s "$file" "$R/orig/$(basename "$file")"; then echo "$name NOT-APPLIED" | tee -a "$OUT/summary.txt"; return; fi
  diff "$R/orig/$(basename "$file")" "$file" > "$OUT/$name.diff"
  if ! dotnet build "$project" -c Debug > "$OUT/$name.build.log" 2>&1; then echo "$name BUILD-FAILED" | tee -a "$OUT/summary.txt"; return; fi
  dotnet test "$project" -c Debug --no-build --no-restore --filter "$filter" > "$OUT/$name.test.log" 2>&1
  local rc=$?
  local line; line=$(grep -E "Passed!|Failed!" "$OUT/$name.test.log" | tail -1)
  local failed; failed=$(grep -E "^\s+Failed " "$OUT/$name.test.log" | sed -E 's/^\s+Failed ([^ ]+).*/\1/' | sort -u | head -8 | tr '\n' ' ')
  if [ $rc -ne 0 ]; then echo "$name KILLED rc=$rc | $line | $failed" | tee -a "$OUT/summary.txt"; else echo "$name SURVIVED | $line" | tee -a "$OUT/summary.txt"; fi
}
T=CuriousContraptions.tests/CuriousContraptions.tests.csproj
B=tools/BundleElements.Tests/BundleElements.Tests.csproj
GATE="FullyQualifiedName~DeclaredPrismaticDriveTests|FullyQualifiedName~LatchSupplyFixtureTests|FullyQualifiedName~DeclaredPresentationTests|FullyQualifiedName~PoweredGateTests|FullyQualifiedName~SliderInstanceTests|FullyQualifiedName~PrismaticConstraintTests|FullyQualifiedName~BothGateTests"
run R1-static-velocity-kept CuriousContraptions.Simulation/WorkshopPhysicsEngine.cs 's/\n\s*b\.Vx = b\.Vy = b\.Vz = b\.Wx = b\.Wy = b\.Wz = 0f;//' $T "$GATE"
run R2-blocked-no-contact engine/gpu/PhysicsPresentationRead.cs 's/public bool Blocked => MovingContact && /public bool Blocked => /' $T "$GATE"
run R3-power-cap-off CuriousContraptions.Simulation/PrismaticConstraintSolver.cs 's/return power > drive\.MaxPower \? MathF\.CopySign\(drive\.MaxPower \/ MathF\.Abs\(speed\), force\) : force;/return force;/' $T "$GATE"
run R4-contact-last-substep-only CuriousContraptions.Simulation/WorkshopPhysicsEngine.cs 's/(\/\/ The committed moving-contact lane)/joint.MovingContact = false;\n                $1/' $T "$GATE"
run R5-base-contact-counted CuriousContraptions.Simulation/WorkshopPhysicsEngine.cs 's/c\.B != joint\.A\)/true)/; s/c\.A != joint\.A\)\)/true))/' $T "$GATE"
run R6-validator-target-unbounded CuriousContraptions.Bundle/BundleValidator.cs 's/drive\.Target, parameters, slider\.LowerLimit, slider\.UpperLimit/drive.Target, parameters, float.MinValue, float.MaxValue/' $B "FullyQualifiedName~BundleValidatorTests"
run R7-undriven-consumer-admitted engine/gpu/SupplyRoute.cs 's/ \|\| !supplied \|\|/ ||/' $T "$GATE"
run R8-admission-target-unbounded engine/gpu/PrismaticConstraint.cs 's/Target < joint\.Lower \|\| Target > joint\.Upper \|\|//' $T "$GATE"
run R9-weight-term-dropped CuriousContraptions.Simulation/PrismaticConstraintSolver.cs 's/var weight = -mass \* gravityAlongAxis;/var weight = 0f;/' $T "$GATE"
run R10-holding-term-dropped CuriousContraptions.Simulation/PrismaticConstraintSolver.cs 's/var holding = stiffness \* coordinate;/var holding = 0f;/' $T "$GATE"
run R11-blocked-speed-legacy engine/gpu/PhysicsPresentationRead.cs 's/public const float RestSpeed = \.01f;/public const float RestSpeed = 1e-5f;/' $T "$GATE"
restore
dotnet build $T -c Debug > "$OUT/restore.build.log" 2>&1; echo "restored build rc=$?" | tee -a "$OUT/summary.txt"
