using System;
using System.Collections.Generic;
using System.IO;
using CuriousContraptions.Gpu;
using Xunit;

namespace CuriousContraptions.Tests;

/// <summary>Reviewer probe (throwaway): every instance kind saved by the pre-conversion (59ecb70b) production encoder.</summary>
public sealed class ReviewOldSaveProbe
{
    [Fact]
    public void WriteOldSaves()
    {
        var dir = Environment.GetEnvironmentVariable("REVIEW_OLD_SAVES") ?? throw new InvalidOperationException("REVIEW_OLD_SAVES unset");
        Directory.CreateDirectory(dir);
        var kinds = new List<(string Name, IWorkshopInstance Instance)>
        {
            ("spring", WorkshopInput.Springboard(new(1), 0, 1, 0, 0, 0, 0, 1, SpringboardSettings.Default)),
            ("spring_custom", WorkshopInput.Springboard(new(1), 0, 1, 0, 0, 0, 0, 1, new(250f, .5f))),
            ("battery", WorkshopInput.Battery(new(1), 0, 1, 0, 0, 0, 0, 1, ElectricalSourceSettings.Default)),
            ("battery_custom", WorkshopInput.Battery(new(1), 0, 1, 0, 0, 0, 0, 1, new(new(60f), new(30f), .5f, ElectricalEnable.Disabled))),
            ("bumper", WorkshopInput.Bumper(new(1), 0, 1, 0, 0, 0, 0, 1, BumperWork.Default)),
            ("domino", WorkshopInput.Domino(new(1), 0, 1, 0, 0, 0, 0, 1)),
            ("lamp", WorkshopInput.Lamp(new(1), 0, 1, 0, 0, 0, 0, 1)),
            ("latch", WorkshopInput.Latch(new(1), 0, 1, 0, 0, 0, 0, 1)),
            ("ball_detector", WorkshopInput.BallDetector(new(1), 0, 1, 0, 0, 0, 0, 1)),
            ("pressure_plate", WorkshopInput.PressurePlate(new(1), 0, 1, 0, 0, 0, 0, 1, 2f)),
            ("counter", WorkshopInput.Counter(new(1), 0, 1, 0, 0, 0, 0, 1, 3)),
            ("clock", WorkshopInput.Clock(new(1), 0, 1, 0, 0, 0, 0, 1, 1f)),
            ("pipe", WorkshopInput.Pipe(new(1), 0, 1, 0, 0, 0, 0, 1)),
            ("pipe_bend_45", WorkshopInput.PipeBend45(new(1), 0, 1, 0, 0, 0, 0, 1)),
            ("funnel", WorkshopInput.Funnel(new(1), 0, 1, 0, 0, 0, 0, 1)),
            ("ramp_default", WorkshopInput.Generic(PartId.Ramp, new(1), 0, 1, 0, 0, 0, 0, 1)),
            ("basketball", WorkshopInput.Basketball(new(1), 0, 1, 0, 0, 0, 0, 1)),
            ("bowling", WorkshopInput.Ball(PartId.BowlingBall, new(1), 0, 1, 0, 0, 0, 0, 1)),
        };
        foreach (var (name, instance) in kinds)
        {
            var save = new WorkshopSavedConstruction(new(new(1), WorkshopCadenceSettings.Default(), new([instance])), new(2));
            File.WriteAllBytes(Path.Combine(dir, name + ".bin"), WorkshopSaveCodec.Encode(save));
        }
    }
}
