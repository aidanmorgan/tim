using CuriousContraptions.Gpu;

namespace CuriousContraptions.Tests;

/// <summary>Reviewer probe (throwaway): the candidate's exact CuriousContraptions.Simulation WarmupSimulation construction (battery,
/// pressure plate, powered gate, two electrical wires, no dynamic body) validates, compiles, runs 5 real ticks and resets exactly.</summary>
public sealed class ReviewWarmupProbe
{
    [Fact]
    public async Task TheWarmupConstructionRunsAndResets()
    {
        var battery = WorkshopInput.Battery(new(1), -2, 0, 0, 0, 0, 0, 1, ElectricalSourceSettings.Default);
        var plate = WorkshopInput.PressurePlate(new(2), 0, 0, 0, 0, 0, 0, 1, WorkshopPressurePlate.PlacedMinimumMass);
        var gate = WorkshopInput.Generic(PartId.PoweredGate, new(3), 2, 0, 0, 0, 0, 0, 1);
        var wire1 = new WorkshopConnection(battery.Id, WorkshopSocket.Supply, plate.Id, WorkshopSocket.PowerIn, WorkshopConnectionDomain.Electrical);
        var wire2 = new WorkshopConnection(plate.Id, WorkshopSocket.Supply, gate.Id, WorkshopSocket.PowerIn, WorkshopConnectionDomain.Electrical);
        var construction = new WorkshopConstruction(new(2), WorkshopCadenceSettings.Default(), new([battery, plate, gate]),
            Connections: new([wire1, wire2]));
        construction.Validate();
        var scene = WorkshopPhysicsCompiler.Compile(construction, new(1, 2));
        Assert.True(scene.Bodies.Length >= 3, $"compiled bodies {scene.Bodies.Length}"); // the powered gate declares more than one body
        var (built, run, reset) = await NoFixedCapacityTests.RunAndReset(construction, 5);
        Assert.Equal(5, run.Count);
        NoFixedCapacityTests.AssertSameState(built, reset, "warmup Reset"); // the suite's own exact-Reset comparison (world epoch aside)
    }
}
