using System.Numerics;
using CuriousContraptions.Gpu;
using CuriousContraptions.Simulation;

namespace CuriousContraptions.Tests;

/// <summary>Reviewer probes (engine restitution slice, throwaway worktree only).</summary>
public sealed class ReviewRestitutionProbeTests(ITestOutputHelper output)
{
    private static readonly float BenchY = LawScene.BenchY;
    private static readonly float G = -(float)(Half)LawScene.Gravity;
    private static float H(float v) => (float)(Half)v;
    private static Vector3 At(float x, float y, float z = 0f) => new(x, y, z);
    private static double Sq(Vector3 v) => (double)v.X * v.X + (double)v.Y * v.Y + (double)v.Z * v.Z;
    private static double Slop => G * WorkshopConstraintSolver.ContactSlop;

    private static readonly string[] Kinds = ["spinSphere", "box", "cylinder", "boxOnDynBox", "sphereOnSphere", "sphereOnBoxEdge", "cylOnCyl", "rollingSphere"];
    private enum Shape { Sphere, Box, Cylinder }
    private sealed record Dyn(GpuBodyId Id, Shape Shape, float Mass, float R, Vector3 Half, bool Gravity);

    private static double Energy(Dyn d, LawBody b)
    {
        var w = Vector3.Transform(b.Angular, Quaternion.Inverse(b.Rotation));
        double m = H(d.Mass);
        double rot = d.Shape switch
        {
            Shape.Sphere => .4 * m * d.R * d.R * Sq(w),
            Shape.Cylinder => m * (d.R * d.R / 4.0 + d.Half.Y * d.Half.Y / 3.0) * ((double)w.X * w.X + (double)w.Z * w.Z) + m * d.R * d.R / 2.0 * w.Y * w.Y,
            _ => m * (((double)d.Half.Y * d.Half.Y + d.Half.Z * d.Half.Z) / 3 * w.X * w.X + ((double)d.Half.X * d.Half.X + d.Half.Z * d.Half.Z) / 3 * w.Y * w.Y +
                ((double)d.Half.X * d.Half.X + d.Half.Y * d.Half.Y) / 3 * w.Z * w.Z),
        };
        return .5 * m * Sq(b.Velocity) + .5 * rot + (d.Gravity ? m * G * (b.Position.Y - BenchY) : 0);
    }

    private static (double Worst, int Tick) Run(LawWorld world, List<Dyn> bodies, int ticks)
    {
        double? prev = null; var worst = double.NegativeInfinity; var at = -1;
        world.Step(ticks, (t, s) =>
        {
            var e = bodies.Sum(d => Energy(d, s[d.Id]));
            if (prev is { } p && e - p > worst) { worst = e - p; at = t; }
            prev = e;
        });
        return (worst, at);
    }

    private static Quaternion RandQ(Random r) => Quaternion.Normalize(new((float)(r.NextDouble() * 2 - 1), (float)(r.NextDouble() * 2 - 1),
        (float)(r.NextDouble() * 2 - 1), (float)(r.NextDouble() * 2 - 1)));
    private static Vector3 RandV(Random r, float k) => new((float)(r.NextDouble() * 2 - 1) * k, (float)(r.NextDouble() * 2 - 1) * k, (float)(r.NextDouble() * 2 - 1) * k);

    /// <summary>Held-back seeds: single and multi-body random poses incl. fast-spinning spheres, dynamic box/box, sphere/sphere.</summary>
    [Theory]
    [InlineData(0f)]
    [InlineData(.3f)]
    [InlineData(.55f)]
    [InlineData(1f)]
    public void HeldBackRandomEnergy(float e)
    {
        var mat = new LawMaterial(e, .1f, .5f, 0f);
        var worstOver = new List<string>();
        foreach (var kind in Kinds)
        {
            var rng = new Random(880_000 + Array.IndexOf(Kinds, kind) * 1000 + (int)(e * 100));
            double worst = double.NegativeInfinity; int worstCase = -1, worstTick = -1; double massSum = 1;
            for (var i = 0; i < 30; i++)
            {
                var scene = new LawScene();
                var bodies = new List<Dyn>();
                var q = RandQ(rng); var h = (float)(rng.NextDouble() * .6 + .2);
                switch (kind)
                {
                    case "spinSphere":
                        bodies.Add(new(scene.Sphere(At(0, BenchY + .1f + h), .1f, 1f, mat, RandV(rng, 1), RandV(rng, 20), 0f, LawScene.Gravity, rotation: q), Shape.Sphere, 1f, H(.1f), default, true));
                        break;
                    case "rollingSphere":
                        bodies.Add(new(scene.Sphere(At(0, BenchY + .1f + h * .1f), .1f, 1f, mat with { Rolling = .05f }, At(3, -1, 1), At(0, 5, -30), 0f, LawScene.Gravity), Shape.Sphere, 1f, H(.1f), default, true));
                        break;
                    case "box":
                        bodies.Add(new(scene.Box(At(0, BenchY + .3f + h), At(.25f, .1f, .15f), 1f, mat, q, RandV(rng, 1), RandV(rng, 6)), Shape.Box, 1f, 0, At(H(.25f), H(.1f), H(.15f)), true));
                        break;
                    case "cylinder":
                        bodies.Add(new(scene.Cylinder(At(0, BenchY + .3f + h), .2f, .15f, 1f, mat, q, RandV(rng, 1), RandV(rng, 6)), Shape.Cylinder, 1f, H(.2f), At(0, H(.15f), 0), true));
                        break;
                    case "boxOnDynBox":
                        bodies.Add(new(scene.Box(At(0, BenchY + .1f), At(.3f, .1f, .3f), 2f, mat), Shape.Box, 2f, 0, At(H(.3f), H(.1f), H(.3f)), true));
                        bodies.Add(new(scene.Box(At((float)rng.NextDouble() * .2f, BenchY + .5f + h, 0), At(.25f, .1f, .15f), 1f, mat, q, default, RandV(rng, 4)), Shape.Box, 1f, 0, At(H(.25f), H(.1f), H(.15f)), true));
                        break;
                    case "sphereOnSphere":
                        bodies.Add(new(scene.Sphere(At(0, BenchY + .2f), .2f, 3f, mat, default, default, 0f, LawScene.Gravity), Shape.Sphere, 3f, H(.2f), default, true));
                        bodies.Add(new(scene.Sphere(At((float)rng.NextDouble() * .3f - .15f, BenchY + .6f + h, (float)rng.NextDouble() * .2f - .1f), .1f, 1f, mat, RandV(rng, .5f), RandV(rng, 10), 0f, LawScene.Gravity), Shape.Sphere, 1f, H(.1f), default, true));
                        break;
                    case "sphereOnBoxEdge":
                        scene.StaticBox(At(0, BenchY + .2f), At(.3f, .2f, .3f), RandQ(rng), mat);
                        bodies.Add(new(scene.Sphere(At((float)rng.NextDouble() * .6f - .3f, BenchY + .9f + h, (float)rng.NextDouble() * .6f - .3f), .1f, 1f, mat, default, RandV(rng, 10), 0f, LawScene.Gravity), Shape.Sphere, 1f, H(.1f), default, true));
                        break;
                    case "cylOnCyl":
                        scene.StaticCylinder(At(0, BenchY + .3f), .3f, .5f, Quaternion.CreateFromAxisAngle(Vector3.UnitX, MathF.PI / 2), mat);
                        bodies.Add(new(scene.Cylinder(At((float)rng.NextDouble() * .4f - .2f, BenchY + .9f + h), .2f, .15f, 1f, mat, q, default, RandV(rng, 6)), Shape.Cylinder, 1f, H(.2f), At(0, H(.15f), 0), true));
                        break;
                }
                massSum = bodies.Sum(x => (double)H(x.Mass));
                var world = scene.Admit();
                var (w, t) = Run(world, bodies, 360);
                if (w > worst) { worst = w; worstCase = i; worstTick = t; }
            }
            var line = $"e={e} {kind}: worst rise {worst * 1000:F4} mJ (case {worstCase}, tick {worstTick}), bound {massSum * Slop * 1000:F3} mJ";
            output.WriteLine(line);
            if (worst > massSum * Slop) worstOver.Add(line);
        }
        Assert.Empty(worstOver);
    }

    /// <summary>Phase sweep: a ball approaching a face at speed v from 64 start offsets across two substeps of travel; no gravity.
    /// Every outgoing speed must equal e·v (no missed or double bounce, no phase dependence).</summary>
    [Theory]
    [InlineData(1f, .5f)]
    [InlineData(4f, .5f)]
    [InlineData(20f, .5f)]
    [InlineData(60f, .5f)]
    [InlineData(4f, 1f)]
    [InlineData(60f, 1f)]
    public void PhaseSweepBounceRatio(float v, float e)
    {
        var h = 1f / 480;
        double lo = double.MaxValue, hi = double.MinValue;
        for (var k = 0; k < 64; k++)
        {
            var scene = new LawScene(bench: false);
            scene.StaticBox(At(0, 0), At(1, .5f, 1), null, new LawMaterial(1f, .1f, 0f, 0f));
            var start = .5f + .1f + .002f + v * h * 8 + v * h * 2 * k / 64f;
            var ball = scene.Sphere(At(0, start), .1f, 1f, new LawMaterial(e, .1f, 0f, 0f), At(0, -v, 0), default, 0f, 0f);
            var world = scene.Admit();
            var final = world.Step(world.TicksPerSecond / 4)[ball];
            var ratio = final.Velocity.Y / v;
            lo = Math.Min(lo, ratio); hi = Math.Max(hi, ratio);
            Assert.True(final.Position.Y > .5f, $"tunnelled at offset {k}");
        }
        output.WriteLine($"v={v} e={e}: ratio [{lo:F5}, {hi:F5}] (expected {H(e) * 1.0:F5})");
        Assert.InRange(lo, H(e) * .99, H(e) * 1.01);
        Assert.InRange(hi, H(e) * .99, H(e) * 1.01);
    }

    /// <summary>Box flat at high speed with a tilt sweep (no gravity): no gain, ratio ≤ e.</summary>
    [Theory]
    [InlineData(10f)]
    [InlineData(40f)]
    public void FastBoxNoGain(float v)
    {
        foreach (var e in new[] { .3f, 1f })
            for (var k = 0; k < 16; k++)
            {
                var scene = new LawScene(bench: false);
                scene.StaticBox(At(0, 0), At(2, .5f, 2), null, new LawMaterial(1f, .1f, .5f, 0f));
                var q = Quaternion.CreateFromYawPitchRoll(k * .2f, k * .05f, k * .03f);
                var id = scene.Box(At(0, 1.2f), At(.25f, .1f, .15f), 1f, new LawMaterial(e, .1f, .5f, 0f), q, At(0, -v, 0), default, 0f);
                var d = new Dyn(id, Shape.Box, 1f, 0, At(H(.25f), H(.1f), H(.15f)), false);
                var world = scene.Admit();
                var e0 = Energy(d, world.Initial(id));
                var max = 0.0;
                world.Step(world.TicksPerSecond / 2, (_, s) => max = Math.Max(max, Energy(d, s[id]) - e0));
                output.WriteLine($"v={v} e={e} k={k}: max gain over initial {max * 1000:F4} mJ");
                Assert.True(max <= Slop, $"v={v} e={e} k={k}: gain {max * 1000:F3} mJ");
            }
    }

    /// <summary>Moving sphere pair (the D9 exception): oblique elastic frictionless collision; measure the deflection error and
    /// energy/momentum.</summary>
    [Theory]
    [InlineData(3f)]
    [InlineData(6f)]
    [InlineData(12f)]
    public void MovingSpherePairOblique(float v)
    {
        var mat = new LawMaterial(1f, .1f, 0f, 0f);
        var scene = new LawScene(bench: false);
        var a = scene.Sphere(At(-1, .1f), .1f, 1f, mat, At(v, 0, 0), default, 0f, 0f);
        var b = scene.Sphere(At(0, 0), .1f, 1f, mat, default, default, 0f, 0f);
        var world = scene.Admit();
        var f = world.Step(world.TicksPerSecond / 2);
        // Impact parameter 0.1 with R = 0.2: n = (cos 30°, -sin 30°) from b toward a reversed: b leaves along (cos30, -sin30).
        var nb = new Vector3(MathF.Cos(MathF.PI / 6), -.5f, 0);
        var angle = Math.Acos(Math.Clamp(Vector3.Dot(Vector3.Normalize(f[b].Velocity), nb), -1f, 1f)) * 180 / Math.PI;
        var ke = .5 * (Sq(f[a].Velocity) + Sq(f[b].Velocity)) + .5 * .4 * H(.1f) * H(.1f) * (Sq(f[a].Angular) + Sq(f[b].Angular));
        output.WriteLine($"v={v}: b dir err {angle:F3}°, KE ratio {ke / (.5 * v * v):F6}, p=({f[a].Velocity + f[b].Velocity})");
        Assert.True(ke <= .5 * v * v * 1.0001 + Slop);
    }

    /// <summary>e = 1 long bouncing (10 s): total energy never exceeds the initial plus slop for a sphere, a flat box, a tilted box.</summary>
    [Fact]
    public void ElasticLongRunNoNetGain()
    {
        var mat = new LawMaterial(1f, .1f, .5f, 0f);
        foreach (var kind in new[] { 0, 1, 2 })
        {
            var scene = new LawScene();
            Dyn d = kind switch
            {
                0 => new(scene.Sphere(At(0, BenchY + .6f), .1f, 1f, mat, default, At(0, 0, 5), 0f, LawScene.Gravity), Shape.Sphere, 1f, H(.1f), default, true),
                1 => new(scene.Box(At(0, BenchY + .6f), At(.25f, .1f, .15f), 1f, mat), Shape.Box, 1f, 0, At(H(.25f), H(.1f), H(.15f)), true),
                _ => new(scene.Box(At(0, BenchY + .6f), At(.25f, .1f, .15f), 1f, mat, Quaternion.CreateFromYawPitchRoll(.4f, .3f, .2f)), Shape.Box, 1f, 0, At(H(.25f), H(.1f), H(.15f)), true),
            };
            var world = scene.Admit();
            var e0 = Energy(d, world.Initial(d.Id));
            var max = double.NegativeInfinity;
            world.Step(10 * world.TicksPerSecond, (_, s) => max = Math.Max(max, Energy(d, s[d.Id]) - e0));
            output.WriteLine($"kind {kind}: max E − E0 over 10 s {max * 1000:F4} mJ");
            Assert.True(max <= Slop, $"kind {kind}: {max * 1000:F3} mJ");
        }
    }

    /// <summary>Dynamic stack detail: a 1 kg box dropped onto a 2 kg box resting on the bench, 60 seeded poses per e.
    /// Reports worst per-tick rise, worst genuine gain (E(t) − max of the 3 preceding ticks) and worst E(t) − E0.</summary>
    [Theory]
    [InlineData(.3f, 1f)]
    [InlineData(.55f, 1f)]
    [InlineData(.55f, .55f)]
    [InlineData(.74f, 1f)]
    [InlineData(1f, 1f)]
    [InlineData(1f, .3f)]
    public void DynamicStackDetail(float e, float bench)
    {
        var mat = new LawMaterial(e, .1f, .5f, 0f);
        double worstRise = double.NegativeInfinity, worstGain = double.NegativeInfinity, worstNet = double.NegativeInfinity;
        int rc = -1, gc = -1, nc = -1, rt = -1;
        for (var i = 0; i < 60; i++)
        {
            var rng = new Random(990_000 + i);
            var scene = new LawScene(bench: false);
            scene.StaticBox(At(0, BenchY - .5f), At(3, .5f, 3), null, new LawMaterial(bench, .1f, .3f, 0f));
            var q = RandQ(rng); var h = (float)(rng.NextDouble() * .6 + .2);
            var bodies = new List<Dyn>
            {
                new(scene.Box(At(0, BenchY + .1f), At(.3f, .1f, .3f), 2f, mat), Shape.Box, 2f, 0, At(H(.3f), H(.1f), H(.3f)), true),
                new(scene.Box(At((float)rng.NextDouble() * .2f, BenchY + .5f + h, 0), At(.25f, .1f, .15f), 1f, mat, q, default, RandV(rng, 4)), Shape.Box, 1f, 0, At(H(.25f), H(.1f), H(.15f)), true),
            };
            var world = scene.Admit();
            var e0 = bodies.Sum(d => Energy(d, world.Initial(d.Id)));
            var hist = new List<double>();
            world.Step(360, (t, s) =>
            {
                var en = bodies.Sum(d => Energy(d, s[d.Id]));
                if (hist.Count > 0 && en - hist[^1] > worstRise) { worstRise = en - hist[^1]; rc = i; rt = t; }
                if (hist.Count >= 3) { var g = en - hist.TakeLast(3).Max(); if (g > worstGain) { worstGain = g; gc = i; } }
                if (en - e0 > worstNet) { worstNet = en - e0; nc = i; }
                hist.Add(en);
            });
        }
        output.WriteLine($"e={e} bench={bench}: rise {worstRise * 1000:F3} mJ (case {rc} tick {rt}); genuine {worstGain * 1000:F3} mJ (case {gc}); net over E0 {worstNet * 1000:F3} mJ (case {nc}); bound {3 * Slop * 1000:F3}");
    }

    private const int TraceCase = 58, TraceTick = 81;

    /// <summary>The held-back e = 1 dynamic stack case (boxOnDynBox) traced per tick, and its genuine gain; e sweep.</summary>
    [Theory]
    [InlineData(.3f)]
    [InlineData(.55f)]
    [InlineData(.85f)]
    [InlineData(.9f)]
    [InlineData(.95f)]
    [InlineData(1f)]
    public void BoxOnDynBoxSweep(float e)
    {
        var mat = new LawMaterial(e, .1f, .5f, 0f);
        var rng = new Random(880_000 + 3000 + (int)(e * 100));
        double worstRise = double.NegativeInfinity, worstGain = double.NegativeInfinity; int rc = -1, gc = -1, rt = -1, gt = -1;
        for (var i = 0; i < 200; i++)
        {
            var scene = new LawScene();
            var q = RandQ(rng); var h = (float)(rng.NextDouble() * .6 + .2);
            var bodies = new List<Dyn>
            {
                new(scene.Box(At(0, BenchY + .1f), At(.3f, .1f, .3f), 2f, mat), Shape.Box, 2f, 0, At(H(.3f), H(.1f), H(.3f)), true),
                new(scene.Box(At((float)rng.NextDouble() * .2f, BenchY + .5f + h, 0), At(.25f, .1f, .15f), 1f, mat, q, default, RandV(rng, 4)), Shape.Box, 1f, 0, At(H(.25f), H(.1f), H(.15f)), true),
            };
            var world = scene.Admit();
            var hist = new List<double>();
            var lines = new List<string>();
            world.Step(360, (t, s) =>
            {
                var en = bodies.Sum(d => Energy(d, s[d.Id]));
                if (hist.Count > 0 && en - hist[^1] > worstRise) { worstRise = en - hist[^1]; rc = i; rt = t; }
                if (hist.Count >= 3) { var g = en - hist.TakeLast(3).Max(); if (g > worstGain) { worstGain = g; gc = i; gt = t; } }
                hist.Add(en);
                if (e == .95f && i == TraceCase && t >= TraceTick - 6 && t <= TraceTick + 3)
                    lines.Add($"  t{t} E={en * 1000:F1} mJ lower v={s[bodies[0].Id].Velocity} w={s[bodies[0].Id].Angular} y={s[bodies[0].Id].Position.Y - BenchY:F5} | top v={s[bodies[1].Id].Velocity} w={s[bodies[1].Id].Angular} y={s[bodies[1].Id].Position.Y - BenchY:F4}");
            });
            foreach (var l in lines) output.WriteLine(l);
        }
        output.WriteLine($"e={e}: rise {worstRise * 1000:F3} mJ (case {rc} t{rt}); genuine {worstGain * 1000:F3} mJ (case {gc} t{gt}); bound {3 * Slop * 1000:F3}");
    }

    /// <summary>Prints the explicit parameters of sweep cases (e, case) so they can be pinned as standalone reproducers.</summary>
    [Theory]
    [InlineData(.95f, 58)]
    [InlineData(1f, 122)]
    public void PrintSweepCase(float e, int target)
    {
        var rng = new Random(880_000 + 3000 + (int)(e * 100));
        for (var i = 0; i <= target; i++)
        {
            var q = RandQ(rng); var h = (float)(rng.NextDouble() * .6 + .2);
            var x = (float)rng.NextDouble() * .2f;
            var spin = RandV(rng, 4);
            if (i == target)
                output.WriteLine(FormattableString.Invariant($"e={e} case {i}: q=({q.X:R}f, {q.Y:R}f, {q.Z:R}f, {q.W:R}f) h={h:R}f x={x:R}f spin=({spin.X:R}f, {spin.Y:R}f, {spin.Z:R}f)"));
        }
    }

    /// <summary>Standalone reproducer: a 1 kg box (half 0.25, 0.1, 0.15; restitution e, threshold 0.1, μ 0.5) dropped onto a 2 kg box
    /// (half 0.3, 0.1, 0.3; same material) resting on the bench plane (Bench material, e 1). Explicit pose, height, offset, spin.</summary>
    [Theory]
    [InlineData(.95f, .15044264f, -.12781543f, .7329633f, .6509954f, .7497563f, .06098506f, -3.4555395f, .9491693f, .3111048f)]
    [InlineData(1f, .5394204f, .28104627f, -.5952043f, .5251385f, .5262131f, .06089176f, 1.5234644f, -.7419849f, -3.30802f)]
    public void DynamicStackReproducer(float e, float qx, float qy, float qz, float qw, float h, float x, float sx, float sy, float sz)
    {
        var mat = new LawMaterial(e, .1f, .5f, 0f);
        var scene = new LawScene();
        var bodies = new List<Dyn>
        {
            new(scene.Box(At(0, BenchY + .1f), At(.3f, .1f, .3f), 2f, mat), Shape.Box, 2f, 0, At(H(.3f), H(.1f), H(.3f)), true),
            new(scene.Box(At(x, BenchY + .5f + h, 0), At(.25f, .1f, .15f), 1f, mat, new Quaternion(qx, qy, qz, qw), default, new(sx, sy, sz)), Shape.Box, 1f, 0, At(H(.25f), H(.1f), H(.15f)), true),
        };
        var world = scene.Admit();
        var hist = new List<double>();
        double rise = double.NegativeInfinity, gain = double.NegativeInfinity; int rt = -1, gt = -1;
        world.Step(360, (t, s) =>
        {
            var en = bodies.Sum(d => Energy(d, s[d.Id]));
            if (hist.Count > 0 && en - hist[^1] > rise) { rise = en - hist[^1]; rt = t; }
            if (hist.Count >= 3) { var g = en - hist.TakeLast(3).Max(); if (g > gain) { gain = g; gt = t; } }
            hist.Add(en);
        });
        output.WriteLine($"reproducer e={e}: worst rise {rise * 1000:F3} mJ (t{rt}); worst genuine gain {gain * 1000:F3} mJ (t{gt}); bound {3 * Slop * 1000:F3} mJ");
        Assert.True(rise <= 3 * Slop, $"rise {rise * 1000:F3} mJ at t{rt}");
    }
