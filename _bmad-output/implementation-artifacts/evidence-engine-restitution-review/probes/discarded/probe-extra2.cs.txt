
    /// <summary>The held-back e = 1 dynamic stack case (boxOnDynBox) traced per tick, and its genuine gain; e sweep.</summary>
    [Theory]
    [InlineData(.85f)]
    [InlineData(.9f)]
    [InlineData(.95f)]
    [InlineData(1f)]
    public void BoxOnDynBoxSweep(float e)
    {
        var mat = new LawMaterial(e, .1f, .5f, 0f);
        var rng = new Random(880_000 + "boxOnDynBox".GetHashCode() % 1000 + (int)(e * 100));
        double worstRise = double.NegativeInfinity, worstGain = double.NegativeInfinity; int rc = -1, gc = -1, rt = -1, gt = -1;
        for (var i = 0; i < 30; i++)
        {
            var scene = new LawScene();
            var q = RandQ(rng); var h = (float)(rng.NextDouble() * .6 + .2);
            var bodies = new List<Dyn>
            {
                new(scene.Box(At(0, BenchY + .1f), At(.3f, .1f, .3f), 2f, mat), Shape.Box, 2f, 0, At(H(.3f), H(.1f), H(.3f)), true),
                new(scene.Box(At((float)rng.NextDouble() * .2f, BenchY + .5f + h, 0), At(.25f, .1f, .15f), 1f, mat, q, default, RandV(rng, 4)), Shape.Box, 1f, 0, At(H(.25f), H(.1f), H(.15f)), true),
            };
            var world = scene.Admit();
            var hist = new List<double>();
            var lines = new List<string>();
            world.Step(360, (t, s) =>
            {
                var en = bodies.Sum(d => Energy(d, s[d.Id]));
                if (hist.Count > 0 && en - hist[^1] > worstRise) { worstRise = en - hist[^1]; rc = i; rt = t; }
                if (hist.Count >= 3) { var g = en - hist.TakeLast(3).Max(); if (g > worstGain) { worstGain = g; gc = i; gt = t; } }
                hist.Add(en);
                if (e == 1f && i == 22 && t >= 36 && t <= 50)
                    lines.Add($"  t{t} E={en * 1000:F1} mJ lower v={s[bodies[0].Id].Velocity} w={s[bodies[0].Id].Angular} y={s[bodies[0].Id].Position.Y - BenchY:F5} | top v={s[bodies[1].Id].Velocity} w={s[bodies[1].Id].Angular} y={s[bodies[1].Id].Position.Y - BenchY:F4}");
            });
            foreach (var l in lines) output.WriteLine(l);
        }
        output.WriteLine($"e={e}: rise {worstRise * 1000:F3} mJ (case {rc} t{rt}); genuine {worstGain * 1000:F3} mJ (case {gc} t{gt}); bound {3 * Slop * 1000:F3}");
    }
}
