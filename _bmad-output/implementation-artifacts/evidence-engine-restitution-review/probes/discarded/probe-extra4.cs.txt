
    /// <summary>Standalone reproducer: a 1 kg box (half 0.25, 0.1, 0.15; restitution e, threshold 0.1, μ 0.5) dropped onto a 2 kg box
    /// (half 0.3, 0.1, 0.3; same material) resting on the bench plane (Bench material, e 1). Explicit pose, height, offset, spin.</summary>
    [Theory]
    [InlineData(.95f, .15044264f, -.12781543f, .7329633f, .6509954f, .7497563f, .06098506f, -3.4555395f, .9491693f, .3111048f)]
    [InlineData(1f, .5394204f, .28104627f, -.5952043f, .5251385f, .5262131f, .06089176f, 1.5234644f, -.7419849f, -3.30802f)]
    public void DynamicStackReproducer(float e, float qx, float qy, float qz, float qw, float h, float x, float sx, float sy, float sz)
    {
        var mat = new LawMaterial(e, .1f, .5f, 0f);
        var scene = new LawScene();
        var bodies = new List<Dyn>
        {
            new(scene.Box(At(0, BenchY + .1f), At(.3f, .1f, .3f), 2f, mat), Shape.Box, 2f, 0, At(H(.3f), H(.1f), H(.3f)), true),
            new(scene.Box(At(x, BenchY + .5f + h, 0), At(.25f, .1f, .15f), 1f, mat, new Quaternion(qx, qy, qz, qw), default, new(sx, sy, sz)), Shape.Box, 1f, 0, At(H(.25f), H(.1f), H(.15f)), true),
        };
        var world = scene.Admit();
        var hist = new List<double>();
        double rise = double.NegativeInfinity, gain = double.NegativeInfinity; int rt = -1, gt = -1;
        world.Step(360, (t, s) =>
        {
            var en = bodies.Sum(d => Energy(d, s[d.Id]));
            if (hist.Count > 0 && en - hist[^1] > rise) { rise = en - hist[^1]; rt = t; }
            if (hist.Count >= 3) { var g = en - hist.TakeLast(3).Max(); if (g > gain) { gain = g; gt = t; } }
            hist.Add(en);
        });
        output.WriteLine($"reproducer e={e}: worst rise {rise * 1000:F3} mJ (t{rt}); worst genuine gain {gain * 1000:F3} mJ (t{gt}); bound {3 * Slop * 1000:F3} mJ");
        Assert.True(rise <= 3 * Slop, $"rise {rise * 1000:F3} mJ at t{rt}");
    }
}
