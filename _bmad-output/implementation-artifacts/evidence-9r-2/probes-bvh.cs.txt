[Theory]
[MemberData(nameof(GoldenScenes))]
public async Task ProbeGolden(string scene)
{
    var reads = await WorkshopPhysicsEngineTests.Run(Bench(Golden(scene)), 240);
    using var hash = System.Security.Cryptography.IncrementalHash.CreateHash(System.Security.Cryptography.HashAlgorithmName.SHA256);
    var word = new byte[4];
    void F(float v) { System.Buffers.Binary.BinaryPrimitives.WriteSingleLittleEndian(word, v); hash.AppendData(word); }
    void I(int v) { System.Buffers.Binary.BinaryPrimitives.WriteInt32LittleEndian(word, v); hash.AppendData(word); }
    foreach (var read in reads)
    {
        for (var i = 0; i < read.Bodies.Count; i++)
        {
            var b = read.Bodies[i];
            I(b.Body.Cell.X); I(b.Body.Cell.Y); I(b.Body.Cell.Z); F(b.Body.Local.X); F(b.Body.Local.Y); F(b.Body.Local.Z);
            F(b.Body.Velocity.X); F(b.Body.Velocity.Y); F(b.Body.Velocity.Z);
            F(b.Rotation.X); F(b.Rotation.Y); F(b.Rotation.Z); F(b.Rotation.W);
            F(b.AngularVelocity.X); F(b.AngularVelocity.Y); F(b.AngularVelocity.Z);
        }
        for (var i = 0; i < read.ContactWorks.Count; i++) F(read.ContactWorks[i].RemainingEnergy.Value);
    }
    throw new InvalidOperationException($"GOLDEN {scene} bodies={reads[^1].Bodies.Count} sha256={Convert.ToHexString(hash.GetHashAndReset())}");
}
