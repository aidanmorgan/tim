#!/bin/bash
# Reviewer: regenerate wire C#/TS with pinned flatc 25.2.10 and diff against the candidate. Run from the worktree root.
S=/private/tmp/claude-501/-Users-aidan-dev-personal-tim/7e56a042-6670-44d4-ae0f-4efd972de00e/scratchpad
F=$S/flatc/flatc
G=$(mktemp -d "$S/review-regen.XXXX")
mkdir -p "$G/cs" "$G/ts"
"$F" --csharp -o "$G/cs" schemas/workshop_wire.fbs
"$F" --ts -o "$G/ts" schemas/workshop_wire.fbs
OUT=$G/cs
{
  printf '%s\n' '// <auto-generated>' '//  automatically generated by the FlatBuffers compiler, do not modify' '// </auto-generated>' '' \
    'namespace CuriousContraptions.Wire' '{' 'using global::System;' 'using global::System.Collections.Generic;' \
    'using global::Google.FlatBuffers;' ''
  first=1
  for f in $(ls "$OUT/CuriousContraptions/Wire" | sort); do
    [ $first -eq 0 ] && echo; first=0
    awk 'started && /^}$/ { last = NR } { lines[NR] = $0 } /^\{$/ && !started { start = NR + 1; started = 1 }
         END { while (lines[start] == "" || lines[start] ~ /^using /) start++;
               end = last - 1; while (end >= start && lines[end] == "") end--; for (i = start; i <= end; i++) print lines[i] }' \
      "$OUT/CuriousContraptions/Wire/$f"
  done
  echo; echo '}'
} > "$G/WorkshopWireGenerated.cs"
OUT=$G/ts
{
  echo '// automatically generated by the FlatBuffers compiler, do not modify'
  echo 'import * as flatbuffers from "flatbuffers";'
  echo
  for name in $(grep -E '^(enum|struct|table) ' schemas/workshop_wire.fbs | awk '{print $2}'); do
    file=$(echo "$name" | sed -E 's/([a-z0-9])([A-Z])/\1-\2/g' | tr 'A-Z' 'a-z')
    grep -v -E '^// automatically generated|^import ' "$OUT/curious-contraptions/wire/$file.ts" | cat -s | sed '1{/^$/d;}'
    echo
  done
} > "$G/generated.ts"
echo "flatc: $("$F" --version), sha256 $(shasum -a 256 "$F" | cut -d' ' -f1); scratch $G"
echo "== C# candidate engine/gpu/WorkshopWireGenerated.cs vs fresh:"
diff engine/gpu/WorkshopWireGenerated.cs "$G/WorkshopWireGenerated.cs"; echo "diff rc=$?"
echo "== TS generated part of tools/e2e/workshop-wire.ts vs fresh:"
M=$(grep -n "hand-written helpers" tools/e2e/workshop-wire.ts | cut -d: -f1)
head -n $((M-1)) tools/e2e/workshop-wire.ts > "$G/cand-generated.ts"
diff "$G/cand-generated.ts" "$G/generated.ts"; echo "diff rc=$?"
echo "== base (HEAD) committed C# vs fresh from base schema:"
mkdir -p "$G/base/schemas"; git show HEAD:schemas/workshop_wire.fbs > "$G/base/schemas/workshop_wire.fbs"
git show HEAD:engine/gpu/WorkshopWireGenerated.cs > "$G/base-committed.cs"
mkdir -p "$G/bcs"; "$F" --csharp -o "$G/bcs" "$G/base/schemas/workshop_wire.fbs"
OUT=$G/bcs
{
  printf '%s\n' '// <auto-generated>' '//  automatically generated by the FlatBuffers compiler, do not modify' '// </auto-generated>' '' \
    'namespace CuriousContraptions.Wire' '{' 'using global::System;' 'using global::System.Collections.Generic;' \
    'using global::Google.FlatBuffers;' ''
  first=1
  for f in $(ls "$OUT/CuriousContraptions/Wire" | sort); do
    [ $first -eq 0 ] && echo; first=0
    awk 'started && /^}$/ { last = NR } { lines[NR] = $0 } /^\{$/ && !started { start = NR + 1; started = 1 }
         END { while (lines[start] == "" || lines[start] ~ /^using /) start++;
               end = last - 1; while (end >= start && lines[end] == "") end--; for (i = start; i <= end; i++) print lines[i] }' \
      "$OUT/CuriousContraptions/Wire/$f"
  done
  echo; echo '}'
} > "$G/base-fresh.cs"
diff "$G/base-committed.cs" "$G/base-fresh.cs"; echo "diff rc=$?"
