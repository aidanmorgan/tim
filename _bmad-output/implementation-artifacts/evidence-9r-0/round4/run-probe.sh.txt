#!/bin/sh
# Story 9R.0 round-3 bypass probe: stage the working tree, attempt a real commit (hooks run), require that it is
# refused with HEAD unmoved and that the log names the expected refusal, then restore the clean candidate tree.
# Usage: run-probe.sh <worktree> <label> <log> <expected-extended-regex> [staged-only]
wt="$1"; label="$2"; log="$3"; expected="$4"; mode="${5:-all}"
before=$(git -C "$wt" rev-parse HEAD)
if [ "$mode" != "staged-only" ]; then git -C "$wt" add -A; fi
{
    echo "probe: $label"
    echo "expected refusal (ERE): $expected"
    echo "head_before: $before"
    echo "--- staged:"
    git -C "$wt" diff --cached --stat
    echo "--- unstaged:"
    git -C "$wt" diff --stat
    echo "--- commit output:"
} > "$log"
set +e
git -C "$wt" commit -q -m "probe: $label (must be refused)" > "$log.commit" 2>&1
status=$?
set -e
after=$(git -C "$wt" rev-parse HEAD)
# Only the commit's own output can name the refusal (never the header above, which quotes the pattern).
if grep -Eq -- "$expected" "$log.commit"; then named=yes; else named=NO; fi
echo "--- refusal lines matching the expected pattern:" >> "$log"
grep -E -- "$expected" "$log.commit" | sort -u | head -5 >> "$log"
echo "--- full commit output:" >> "$log"
cat "$log.commit" >> "$log"
rm -f "$log.commit"
moved=$([ "$before" = "$after" ] && echo no || echo YES)
echo "--- result: commit_exit=$status head_after=$after moved=$moved refusal_named=$named" >> "$log"
echo "probe=$label commit_exit=$status moved=$moved refusal_named=$named"
if [ "$moved" = "YES" ]; then
    echo "BYPASS: $label committed as $after; leaving it for inspection" >&2
    exit 2
fi
git -C "$wt" reset -q --hard HEAD
git -C "$wt" clean -fdq
