#!/bin/sh
# Story 9R.0 bypass probe: commit exactly the current index (no add), report the exit code and whether HEAD moved.
# Usage: try-commit-staged.sh <worktree> <label> <log>
wt="$1"; label="$2"; log="$3"
before=$(git -C "$wt" rev-parse HEAD)
echo "staged diff:" > "$log"
git -C "$wt" diff --cached --stat >> "$log"
echo "unstaged diff:" >> "$log"
git -C "$wt" diff --stat >> "$log"
set +e
git -C "$wt" commit -q -m "probe: $label (must be refused)" >> "$log" 2>&1
status=$?
set -e
after=$(git -C "$wt" rev-parse HEAD)
echo "probe=$label commit_exit=$status head_before=$before head_after=$after moved=$([ "$before" = "$after" ] && echo no || echo YES)"
