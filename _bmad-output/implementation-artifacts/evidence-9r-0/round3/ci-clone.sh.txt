#!/bin/sh
# Story 9R.0 round 3: run pages.yml's build steps as written, in a fresh git clone of the committed branch (CI's
# shape: a git checkout, so the guard lists files through git ls-files, with every manifest required).
# Usage: ci-clone.sh <worktree> <branch> <log>
set -eu
wt="$1"; branch="$2"; log="$3"
parent=$(cd "$(mktemp -d "${TMPDIR:-/tmp}/ci-clone.XXXXXX")" && pwd -P)
workspace="$parent/repo"
git clone -q --branch "$branch" "$wt" "$workspace"
cd "$workspace"
GATE="-noAutoResponse -p:CustomAfterMicrosoftCommonTargets=$workspace/CSharpStaticGate.targets"
GATE_TARGETS="-p:CustomAfterMicrosoftCommonTargets=$workspace/CSharpStaticGate.targets"
start=$(date +%s)
{
  echo "workspace=$workspace head=$(git rev-parse HEAD) git_dir=$(git rev-parse --git-dir)"
  dotnet restore CuriousContraptions.slnx --nologo --disable-parallel $GATE
  dotnet build CuriousContraptions.slnx --nologo --no-restore --no-incremental $GATE
  dotnet build CuriousContraptions.slnx --nologo --no-restore --no-incremental -p:PlaytestDiagnostics=true $GATE
  dotnet restore CuriousContraptions.web/CuriousContraptions.web.csproj --nologo --disable-parallel $GATE
  dotnet build CuriousContraptions.web/CuriousContraptions.web.csproj --nologo --no-restore --no-incremental -p:WasmBuildOnlyAfterPublish=true -p:TwoDogAutoImport=false $GATE
  dotnet build tools/anvil/CheckStagedTests.cs --nologo $GATE_TARGETS
  dotnet build tools/anvil/VerifyPolicy.cs --nologo $GATE_TARGETS
  CSHARP_STATIC_GATE_MANIFESTS=required dotnet test CuriousContraptions.tests --nologo -noAutoResponse
  dotnet publish CuriousContraptions.web -c Release --nologo -p:PlaytestDiagnostics=false
  test -s CuriousContraptions.web/AppBundle/index.html
  test -s CuriousContraptions.web/AppBundle/404.html
  test -s CuriousContraptions.web/AppBundle/godot.wasm
  test -s CuriousContraptions.web/AppBundle/godot.pck
  test -s CuriousContraptions.web/AppBundle/_framework/dotnet.js
  echo "CI_CLONE_OK seconds=$(( $(date +%s) - start ))"
} > "$log" 2>&1
echo "exit=0 workspace=$workspace"
