    [Fact]
    public async Task ProbeP1_OneDominoPlacedBeforeThePlateIsCountedButUnderweight()
    {
        // Control for the sum row: one domino alone stays below 0.5 kg, and at 0.3 kg it is enough (so it is counted, not Empty).
        var domino = StandingOn(1, -4, 0); var gate = Gate(2, 0); var battery = Battery(3, 0);
        var light = Plate(4, -4, 0, .5f);
        var closed = await RunBench(Bench([domino, gate, battery, light], Circuit(battery, light, gate)));
        AssertGateClosedThroughout(closed, light, gate, "one domino before a 0.5 kg plate");
        var sensitive = Plate(4, -4, 0, .3f);
        var opened = await RunBench(Bench([domino, gate, battery, sensitive], Circuit(battery, sensitive, gate)));
        AssertGateOpened(opened, sensitive, gate, "one domino before a 0.3 kg plate");
    }
