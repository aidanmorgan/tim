#!/bin/sh
# Murdoch-21 mutants for Story 9A.14 / 9A.18, in the disposable copy murdoch21-mut (detached at de4d1ea5).
# Each mutant: restore the tree, apply one edit, build + run the whole CuriousContraptions.tests (and BundleElements.Tests where the
# edit is in a package), record the counts and failing test names, restore. A non-zero build exit counts as killed by the build.
W=/Users/aidan/dev/personal/tim/.claude/worktrees/murdoch21-mut
E=/Users/aidan/dev/personal/tim/_bmad-output/implementation-artifacts/evidence-9a-14-review/mutants
cd "$W" || exit 1
# reset --hard also clears the index (a `git checkout <rev> -- path` mutant stages its file).
restore() { git reset -q --hard HEAD; git clean -fdq -- parts engine elements ui CuriousContraptions.tests; rm -rf parts; }
run() {
  name=$1; shift
  out="$E/$name.txt"
  { echo "mutant $name at $(date -u +%FT%TZ) head=$(git rev-parse --short HEAD)"; echo "--- diff"; git diff; git status --porcelain --untracked-files=all; echo "--- build+test"; } > "$out"
  dotnet test CuriousContraptions.tests > "$out.log" 2>&1; code=$?
  grep -E "error CS|error BUNDLE|BUNDLE0[0-9][0-9]|\[FAIL\]|Passed!|Failed!|Build FAILED" "$out.log" | sed 's/^\[xUnit.net [0-9:.]*\] *//' | sort -u | head -40 >> "$out"
  echo "test exit=$code" >> "$out"
  rm -f "$out.log"
  echo "$name exit=$code failed=$(grep -E '\[FAIL\]' "$out" | sed 's/ \[FAIL\]//' | tr '\n' ' ')"
  restore
}
restore
