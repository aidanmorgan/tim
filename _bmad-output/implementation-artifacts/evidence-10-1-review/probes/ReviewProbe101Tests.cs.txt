    private static readonly float BenchY = LawScene.BenchY;
    private static readonly float G = -(float)(Half)LawScene.Gravity;
    private static float H(float v) => (float)(Half)v;
    private static Vector3 At(float x, float y, float z = 0f) => new(x, y, z);
    private static readonly LawMaterial Dead = new(0f, .1f, .5f, 0f);
    private static Quaternion About(Vector3 axis, double degrees) =>
        Quaternion.CreateFromAxisAngle(Vector3.Normalize(axis), (float)(degrees * Math.PI / 180));

    /// <summary>Lowest true point of a cylinder (axis = local +Y of rotation q) below its centre.</summary>
    private static float Lowest(Quaternion q, float r, float h)
    {
        var a = Vector3.Transform(Vector3.UnitY, q);
        var t = MathF.Abs(a.Y);
        return h * t + r * MathF.Sqrt(MathF.Max(0, 1 - t * t));
    }

    private static double Energy(LawBody b, float m, float r, float h)
    {
        // World inertia R diag R^T; ω world frame.
        double ix = m * (r * r / 4 + h * h / 3), iy = m * r * r / 2;
        var local = Vector3.Transform(b.Angular, Quaternion.Inverse(b.Rotation));
        var rot = .5 * (ix * local.X * local.X + iy * local.Y * local.Y + ix * local.Z * local.Z);
        var lin = .5 * m * (b.Velocity.X * (double)b.Velocity.X + b.Velocity.Y * (double)b.Velocity.Y + b.Velocity.Z * (double)b.Velocity.Z);
        return lin + rot + m * G * (b.Position.Y - BenchY);
    }

    /// <summary>A cylinder tipped by `degrees` about a horizontal axis at azimuth `azimuth` (0 = tilt about Z, i.e. toward a U0 sample;
    /// 45 = diagonal between samples), released with its lowest true rim point 0.5 mm above the bench.</summary>
    private (float FinalTilt, double MaxPenetration, double MaxEnergyRise, double MaxSpeed) Tip(float r, float h, double degrees,
        double azimuth, int seconds = 4, Vector3 angular = default, SimulationCadence cadence = SimulationCadence.Hz120, float mass = 1f)
    {
        var horizontal = new Vector3((float)Math.Cos(azimuth * Math.PI / 180), 0, (float)Math.Sin(azimuth * Math.PI / 180));
        var tiltAxis = Vector3.Cross(Vector3.UnitY, horizontal); // tilting about this moves the axis toward `horizontal`
        var q = About(tiltAxis, degrees);
        float rr = H(r), hh = H(h);
        var scene = new LawScene();
        var body = scene.Cylinder(At(0, BenchY + Lowest(q, rr, hh) + .0005f), r, h, mass, Dead, q, angular: angular);
        var world = scene.Admit(cadence);
        double maxPen = 0, maxRise = 0, maxSpeed = 0;
        double? previous = null;
        var final = world.Step(seconds * world.TicksPerSecond, (_, s) =>
        {
            var b = s[body];
            var low = b.Position.Y - Lowest(b.Rotation, rr, hh) - BenchY;
            maxPen = Math.Max(maxPen, -low);
            var e = Energy(b, H(mass), rr, hh);
            if (previous is { } p) maxRise = Math.Max(maxRise, e - p);
            previous = e;
            maxSpeed = Math.Max(maxSpeed, b.Speed);
        })[body];
        return (final.Tilt, maxPen, maxRise, maxSpeed);
    }

    // ---- exact separation oracle: max over unit u of the support-function separation ----

    private static double CylSupport(SimCollider c, Vector3d u)
    {
        var a = new Vector3d(c.U1);
        var t = Math.Abs(u.Dot(a));
        return new Vector3d(c.Position).Dot(u) + c.Hy * t + c.Radius * Math.Sqrt(Math.Max(0, 1 - t * t));
    }

    private static double BoxSupport(SimCollider b, Vector3d u) =>
        new Vector3d(b.Position).Dot(u) + b.Hx * Math.Abs(u.Dot(new(b.U0))) + b.Hy * Math.Abs(u.Dot(new(b.U1))) +
        b.Hz * Math.Abs(u.Dot(new(b.U2)));

    private static double Separation(Func<Vector3d, double> supA, Func<Vector3d, double> supB, Vector3d u) =>
        // gap along u (B beyond A in direction u): min over B of b·u − max over A of a·u = −supB(−u) − supA(u)
        -supB(-u) - supA(u);

    private static double ExactSeparation(Func<Vector3d, double> supA, Func<Vector3d, double> supB)
    {
        const int n = 6000;
        var golden = Math.PI * (3 - Math.Sqrt(5));
        var candidates = new List<(double S, Vector3d U)>();
        for (var i = 0; i < n; i++)
        {
            var y = 1 - 2 * (i + .5) / n;
            var rad = Math.Sqrt(1 - y * y);
            var u = new Vector3d(Math.Cos(golden * i) * rad, y, Math.Sin(golden * i) * rad);
            candidates.Add((Separation(supA, supB, u), u));
        }
        var best = double.NegativeInfinity;
        foreach (var (s0, u0) in candidates.OrderByDescending(c => c.S).Take(8))
        {
            var u = u0; var s = s0; var step = .05;
            var rng = new Random(7);
            while (step > 1e-7)
            {
                var improved = false;
                for (var k = 0; k < 24; k++)
                {
                    var v = (u + step * new Vector3d(rng.NextDouble() * 2 - 1, rng.NextDouble() * 2 - 1, rng.NextDouble() * 2 - 1)).Normalized();
                    var sv = Separation(supA, supB, v);
                    if (sv > s) { s = sv; u = v; improved = true; }
                }
                if (!improved) step *= .5;
            }
            best = Math.Max(best, s);
        }
        return best;
    }

    private static Quaternion RandomRotation(Random rng)
    {
        var q = new Quaternion((float)(rng.NextDouble() * 2 - 1), (float)(rng.NextDouble() * 2 - 1), (float)(rng.NextDouble() * 2 - 1),
            (float)(rng.NextDouble() * 2 - 1));
        return Quaternion.Normalize(q);
    }

    private static SimCollider Make(ShapeKind shape, Vector3 p, Quaternion q, float radius, Vector3 half, int index)
    {
        var body = new SimBody
        {
            Px = p.X, Py = p.Y, Pz = p.Z, Qx = q.X, Qy = q.Y, Qz = q.Z, Qw = q.W, Motion = 1, Mass = 1, InvMass = 1,
            Principal = Quaternion.Identity, InvI = Vector3.One, Q0 = Quaternion.Identity
        };
        var c = new SimCollider { Index = index, Body = body, Shape = shape, Radius = radius, Hx = half.X, Hy = half.Y, Hz = half.Z, Rq = Quaternion.Identity };
        WorkshopCollisionKernels.RefreshCollider(c);
        return c;
    }

    private void Sweep(string label, bool cylinderPartner, int poses, double target, Func<Quaternion, Quaternion, bool>? filter = null)
    {
        var rng = new Random(label.GetHashCode() ^ (int)(target * 1e6));
        const float margin = .005f;
        int falsePositive = 0, missed = 0, negativeGhost = 0, count = 0;
        double worstUnder = 0, worstOver = 0;
        string worstCase = "";
        while (count < poses)
        {
            var qa = RandomRotation(rng); var qb = RandomRotation(rng);
            if (filter != null && !filter(qa, qb)) continue;
            count++;
            var cylA = Make(ShapeKind.Cylinder, Vector3.Zero, qa, .3f, new(0, .2f, 0), 0);
            var dir = new Vector3((float)(rng.NextDouble() * 2 - 1), (float)(rng.NextDouble() * 2 - 1), (float)(rng.NextDouble() * 2 - 1));
            dir = Vector3.Normalize(dir);
            SimCollider PartnerAt(float t) => cylinderPartner
                ? Make(ShapeKind.Cylinder, dir * t, qb, .2f, new(0, .35f, 0), 1)
                : Make(ShapeKind.Box, dir * t, qb, 0, new(.25f, .1f, .15f), 1);
            double Exact(SimCollider b) => cylinderPartner
                ? ExactSeparation(u => CylSupport(cylA, u), u => CylSupport(b, u))
                : ExactSeparation(u => CylSupport(cylA, u), u => BoxSupport(b, u));
            float lo = 0f, hi = 2f;
            for (var it = 0; it < 26; it++)
            {
                var mid = .5f * (lo + hi);
                if (Exact(PartnerAt(mid)) > target) hi = mid; else lo = mid;
            }
            var partner = PartnerAt(.5f * (lo + hi));
            var exact = Exact(partner);
            var contacts = new List<ContactPoint>();
            WorkshopCollisionKernels.GenerateManifold(cylA, partner, margin, contacts);
            if (contacts.Count == 0)
            {
                if (exact <= margin - 1e-4) missed++;
                continue;
            }
            var minGap = contacts.Min(c => c.Gap);
            if (exact > margin + 1e-4) falsePositive++;
            if (exact > 1e-4 && minGap < 0) negativeGhost++;
            var err = minGap - exact;
            if (err < worstUnder) { worstUnder = err; worstCase = $"qa {qa} qb {qb} dir {dir} exact {exact:F5} gap {minGap:F5} n={contacts.Count}"; }
            worstOver = Math.Max(worstOver, err);
        }
        output.WriteLine($"{label} target {target * 1000:F1} mm, {poses} poses: false-positive {falsePositive}, missed {missed}, negative-ghost {negativeGhost}, gap−exact in [{worstUnder * 1000:F3}, {worstOver * 1000:F3}] mm");
        if (worstCase.Length > 0) output.WriteLine("  worst: " + worstCase);
    }

    [Fact]
    public void P8_DropEnergyVersusBoxBaseline()
    {
        // Dead material drops at random orientations: per-tick energy rise (should be ~0) vs. a box of similar size.
        var rng = new Random(11);
        double worstCyl = 0, worstBox = 0;
        for (var i = 0; i < 30; i++)
        {
            var q = RandomRotation(rng);
            {
                var scene = new LawScene();
                float r = H(.2f), h = H(.15f);
                var body = scene.Cylinder(At(0, BenchY + .6f), .2f, .15f, 1f, Dead, q);
                var world = scene.Admit();
                double? prev = null;
                world.Step(3 * world.TicksPerSecond, (_, s) =>
                {
                    var e = Energy(s[body], 1f, r, h);
                    if (prev is { } p) worstCyl = Math.Max(worstCyl, e - p);
                    prev = e;
                });
            }
            {
                var scene = new LawScene();
                var body = scene.Box(At(0, BenchY + .6f), At(.2f, .15f, .2f), 1f, Dead, q);
                var world = scene.Admit();
                double? prev = null;
                float hx = H(.2f), hy = H(.15f), hz = H(.2f);
                world.Step(3 * world.TicksPerSecond, (_, s) =>
                {
                    var b = s[body];
                    var local = Vector3.Transform(b.Angular, Quaternion.Inverse(b.Rotation));
                    double ix = (hy * hy + hz * hz) / 3, iy = (hx * hx + hz * hz) / 3, iz = (hx * hx + hy * hy) / 3;
                    var e = .5 * (b.Velocity.LengthSquared()) + .5 * (ix * local.X * local.X + iy * local.Y * local.Y + iz * local.Z * local.Z) +
                        G * (b.Position.Y - BenchY);
                    if (prev is { } p) worstBox = Math.Max(worstBox, e - p);
                    prev = e;
                });
            }
        }
        output.WriteLine($"30 random-orientation dead drops: worst per-tick energy rise cylinder {worstCyl * 1000:F3} mJ, box baseline {worstBox * 1000:F3} mJ");
    }

internal readonly record struct Vector3d(double X, double Y, double Z)
{
    public Vector3d(Vector3 v) : this(v.X, v.Y, v.Z) { }
    public double Dot(Vector3d o) => X * o.X + Y * o.Y + Z * o.Z;
    public static Vector3d operator +(Vector3d a, Vector3d b) => new(a.X + b.X, a.Y + b.Y, a.Z + b.Z);
    public static Vector3d operator -(Vector3d a) => new(-a.X, -a.Y, -a.Z);
    public static Vector3d operator *(double s, Vector3d a) => new(s * a.X, s * a.Y, s * a.Z);
    public Vector3d Normalized() { var l = Math.Sqrt(Dot(this)); return new(X / l, Y / l, Z / l); }
}
