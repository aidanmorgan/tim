using CuriousContraptions.LifecycleDesign;
using System.Text.Json;
enum Probe { WorldFull, UiFull, DrainOld, AwaitOldAck, NewWorld, ThirdGeneration, DuplicateGeneration,
    ZeroGeneration, NegativeEntries, OverLimit, EmptyStart, ClockContainment, QuantizedOverlap,
    QuantizedTouch, QuantizedDisjoint, SamplePrecision, ClockExpiry, HistoryRequired, HistoryControl,
    HistoryExactBoundary, HistoryOverBoundary, InvalidMutation, InvalidPoint }
record Outcome(Probe Probe,bool Pass,string Detail);
static class ReviewProbe {
static readonly List<Outcome> Outcomes=[];
static void Check(Probe p,Action test){try{test();Outcomes.Add(new(p,true,"Passed"));}catch(Exception e){Outcomes.Add(new(p,false,e.Message));}}
static void Require(bool v){if(!v)throw new Exception("Unexpected result");}
static void Reject(Action a){try{a();}catch(ArgumentException){return;}throw new Exception("Accepted invalid input");}
static int Main(){
Check(Probe.WorldFull,()=>Require(!ContractFixture.CanAdmit([new(new(7),1),new(new(8),127)],new(8))));
Check(Probe.UiFull,()=>Require(!ContractFixture.CanAdmit([new(new(7),127),new(new(8),1)],new(7))));
Check(Probe.DrainOld,()=>Require(ContractFixture.CanAdmit([new(new(7),64),new(new(8),1)],new(8))));
Check(Probe.AwaitOldAck,()=>Require(!ContractFixture.CanAdmit([new(new(7),64),new(new(8),1)],new(9))));
Check(Probe.NewWorld,()=>Require(ContractFixture.CanAdmit([new(new(7),64)],new(9))));
Check(Probe.ThirdGeneration,()=>Require(!ContractFixture.CanAdmit([new(new(7),1),new(new(9),1)],new(10))));
Check(Probe.DuplicateGeneration,()=>Reject(()=>ContractFixture.CanAdmit([new(new(7),1),new(new(7),1)],new(8))));
Check(Probe.ZeroGeneration,()=>Reject(()=>ContractFixture.CanAdmit([],new(0))));
Check(Probe.NegativeEntries,()=>Reject(()=>ContractFixture.CanAdmit([new(new(7),-1)],new(8))));
Check(Probe.OverLimit,()=>Reject(()=>ContractFixture.CanAdmit([new(new(7),129)],new(7))));
Check(Probe.EmptyStart,()=>Require(ContractFixture.CanAdmit([],new(1))));
Check(Probe.ClockContainment,()=>{var x=ContractFixture.MapSample(ContractFixture.Calibrate(100,150.2m,150.3m,100.7m,.01m,0),175,.01m);Require(x.Lower==124.77m&&x.Upper==125.43m&&x.Upper>=125.41m);});
Check(Probe.QuantizedOverlap,()=>Require(ContractFixture.Consistent(new(49.98m,50.02m),new(50.01m,50.05m))));
Check(Probe.QuantizedTouch,()=>Require(ContractFixture.Consistent(new(49.98m,50.02m),new(50.02m,50.06m))));
Check(Probe.QuantizedDisjoint,()=>Require(!ContractFixture.Consistent(new(49.98m,50.02m),new(50.020001m,50.060001m))));
Check(Probe.SamplePrecision,()=>Reject(()=>ContractFixture.MapSample(new(49,51),175,.000001m)));
Check(Probe.ClockExpiry,()=>Reject(()=>ContractFixture.Calibrate(1,2,2,1,0,500.000001m)));
Check(Probe.HistoryRequired,()=>Require(ContractFixture.HistoryBytes(44057,117+26*256,117+26*256)==173833));
Check(Probe.HistoryControl,()=>Reject(()=>ContractFixture.HistoryBytes(44057,117+26*256,117+26*842)));
Check(Probe.HistoryExactBoundary,()=>Require(ContractFixture.HistoryBytes(30000,30000,5194)==196606));
Check(Probe.HistoryOverBoundary,()=>Reject(()=>ContractFixture.HistoryBytes(30000,30000,5195)));
Check(Probe.InvalidMutation,()=>Reject(()=>ContractFixture.Expected(FailurePoint.Mutation,(MutationFamily)0)));
Check(Probe.InvalidPoint,()=>Reject(()=>ContractFixture.Expected((FailurePoint)0,MutationFamily.PresentationHistory)));
Console.WriteLine(JsonSerializer.Serialize(new{DesignOnly=true,Count=Outcomes.Count,Outcomes}));return Outcomes.All(x=>x.Pass)?0:1;
}}
