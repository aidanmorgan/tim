using System.Text.Json;
using System.Text.Json.Serialization;

namespace CuriousContraptions.PerformanceTools;

public enum AttemptPurpose { Warmup, Measurement }
public enum AttemptEnd { Timeout, Goal, Interrupted }
public enum CadenceTier { Sixty, Ninety }
public enum PresentationEvidence { CompositorTrace, AnimationFrameCallback, SyntheticRehearsal }
public enum MetricVerdict { Pass, Fail, Incomplete }
public enum TimelinePhase { Active, Reset, SaveLoad, Idle, Hidden }
public readonly record struct AttemptId(Guid Value);
public readonly record struct PresentationId(ulong Value);
public sealed record PresentedFrame(PresentationId Id, decimal Milliseconds);
public sealed record AttemptContract(AttemptEnd ExpectedEnd, int MinimumTicks,
    decimal MinimumWallMilliseconds, int MinimumPresentations, CadenceTier Tier);
public sealed record AttemptObservation(AttemptId Id, AttemptPurpose Purpose, AttemptEnd End,
    decimal StartMilliseconds, decimal EndMilliseconds, int FirstTick, int LastTick,
    bool StartAcknowledged, bool StopObserved, bool Continuous, decimal MappingUncertaintyMilliseconds,
    PresentationEvidence Evidence, PresentedFrame[] Presentations);
public sealed record AttemptMetrics(MetricVerdict Coverage, MetricVerdict ArithmeticPacing,
    MetricVerdict BrowserPacing, MetricVerdict SimulationRate, int DistinctPresentations,
    int ScheduledSlots, int MissedSlots, decimal FramesPerSecond, decimal SimulatedWallRatio,
    decimal P95IntervalMilliseconds, decimal P99IntervalMilliseconds);
public sealed record TimelineInterval(TimelinePhase Phase, decimal StartMilliseconds,
    decimal EndMilliseconds, AttemptId? Attempt);
public sealed record ThermalContract(decimal MinimumDurationMilliseconds,
    decimal MinimumActiveFraction, decimal MaximumGapMilliseconds);
public sealed record ThermalMetrics(MetricVerdict Continuity, MetricVerdict Engagement,
    decimal DurationMilliseconds, decimal ActiveFraction, decimal MaximumGapMilliseconds);
public sealed record MeasurementPacket(AttemptContract Contract, AttemptObservation[] Attempts);

/// <summary>Strict external JSON boundary for arithmetic rehearsal records, not browser certification.</summary>
public static class MeasurementEvidence
{
    private sealed class EndConverter : ExactPlaytestEnumConverter<AttemptEnd>;
    private sealed class PurposeConverter : ExactPlaytestEnumConverter<AttemptPurpose>;
    private sealed class TierConverter : ExactPlaytestEnumConverter<CadenceTier>;
    private sealed class EvidenceConverter : ExactPlaytestEnumConverter<PresentationEvidence>;
    private static JsonSerializerOptions Options()
    {
        var options = new JsonSerializerOptions
        {
            PropertyNamingPolicy = JsonNamingPolicy.CamelCase,
            UnmappedMemberHandling = JsonUnmappedMemberHandling.Disallow,
            RespectRequiredConstructorParameters = true
        };
        options.Converters.Add(new EndConverter());
        options.Converters.Add(new PurposeConverter());
        options.Converters.Add(new TierConverter());
        options.Converters.Add(new EvidenceConverter());
        return options;
    }
    public static MeasurementPacket Read(string json)
    {
        var packet = JsonSerializer.Deserialize<MeasurementPacket>(json, Options())
            ?? throw new InvalidDataException("Null measurement packet.");
        Validate(packet);
        return packet;
    }
    public static string Write(MeasurementPacket packet)
    {
        Validate(packet);
        return JsonSerializer.Serialize(packet, Options());
    }
    private static void Validate(MeasurementPacket packet)
    {
        if (packet.Contract is null || packet.Attempts is null || packet.Attempts.Length == 0)
            throw new InvalidDataException("Missing measurement contract or attempts.");
        var identities = new HashSet<AttemptId>();
        foreach (var attempt in packet.Attempts)
        {
            if (attempt is null || !identities.Add(attempt.Id))
                throw new InvalidDataException("Missing or repeated attempt.");
            PerformanceAudit.AnalyzeAttempt(packet.Contract, attempt);
        }
    }
}

public static partial class PerformanceAudit
{
    public static AttemptMetrics AnalyzeAttempt(AttemptContract contract, AttemptObservation attempt)
    {
        ArgumentNullException.ThrowIfNull(contract);
        ArgumentNullException.ThrowIfNull(attempt);
        if (!Enum.IsDefined(contract.ExpectedEnd) || contract.ExpectedEnd == AttemptEnd.Interrupted ||
            !Enum.IsDefined(contract.Tier) || contract.MinimumTicks <= 0 || contract.MinimumTicks > 3600 ||
            contract.MinimumWallMilliseconds <= 0 || contract.MinimumPresentations < 100 ||
            (contract.ExpectedEnd == AttemptEnd.Timeout && contract.MinimumTicks != 3600))
            throw new InvalidDataException("Invalid frozen attempt contract.");
        if (attempt.Id.Value == Guid.Empty || !Enum.IsDefined(attempt.Purpose) ||
            !Enum.IsDefined(attempt.End) || !Enum.IsDefined(attempt.Evidence) ||
            attempt.StartMilliseconds < 0 || attempt.EndMilliseconds <= attempt.StartMilliseconds ||
            attempt.FirstTick < 0 || attempt.LastTick < attempt.FirstTick || attempt.LastTick > 3600 ||
            attempt.MappingUncertaintyMilliseconds < 0 || attempt.Presentations is null)
            throw new InvalidDataException("Invalid attempt observation.");
        var duration = attempt.EndMilliseconds - attempt.StartMilliseconds;
        var hz = contract.Tier == CadenceTier.Sixty ? 60 : 90;
        // Deadlines are start + n * 1000 / hz. A slot is (previous deadline, deadline].
        // Retain the final partial slot in the denominator; it must not disappear at stop.
        var slots = checked((int)decimal.Ceiling(duration * hz / 1000));
        var occupied = new HashSet<int>();
        var identities = new HashSet<PresentationId>();
        var intervals = new List<decimal>();
        var previousObservation = attempt.StartMilliseconds;
        var previousDistinct = attempt.StartMilliseconds;
        foreach (var frame in attempt.Presentations)
        {
            if (frame is null || frame.Id.Value == 0 ||
                frame.Milliseconds <= previousObservation || frame.Milliseconds > attempt.EndMilliseconds)
                throw new InvalidDataException("Invalid or unordered presentation observation.");
            previousObservation = frame.Milliseconds;
            if (!identities.Add(frame.Id)) continue;
            var slot = checked((int)decimal.Ceiling((frame.Milliseconds - attempt.StartMilliseconds) * hz / 1000)) - 1;
            occupied.Add(slot);
            intervals.Add(frame.Milliseconds - previousDistinct);
            previousDistinct = frame.Milliseconds;
        }
        // Boundary tails are retained for pacing, even if the final presentation precedes stop.
        if (previousDistinct < attempt.EndMilliseconds)
            intervals.Add(attempt.EndMilliseconds - previousDistinct);
        intervals.Sort();
        decimal Percentile(decimal fraction) => intervals.Count == 0 ? duration :
            intervals[(int)decimal.Ceiling(fraction * intervals.Count) - 1];
        var p95 = Percentile(.95m);
        var p99 = Percentile(.99m);
        var fps = identities.Count * 1000m / duration;
        var ratio = (attempt.LastTick - attempt.FirstTick) * 1000m / (120m * duration);
        var coverage = !attempt.StartAcknowledged || !attempt.StopObserved || !attempt.Continuous ||
            attempt.Purpose != AttemptPurpose.Measurement ? MetricVerdict.Incomplete :
            attempt.End != contract.ExpectedEnd ? MetricVerdict.Fail :
            attempt.FirstTick != 0 || attempt.LastTick < contract.MinimumTicks ||
            duration < contract.MinimumWallMilliseconds || identities.Count < contract.MinimumPresentations ?
                MetricVerdict.Incomplete : MetricVerdict.Pass;
        var missed = slots - occupied.Count;
        var pacing = fps >= hz * .99m && missed <= slots * .01m &&
            p95 <= (hz == 60 ? 18m : 12m) && p99 <= (hz == 60 ? 25m : 16.7m) ?
                MetricVerdict.Pass : MetricVerdict.Fail;
        if (coverage != MetricVerdict.Pass) pacing = MetricVerdict.Incomplete;
        // Synthetic values are useful arithmetic oracles, never observed browser presentation.
        // Slot attribution requires an exact common trace clock; mapped uncertainty is incomplete.
        var browser = attempt.Evidence != PresentationEvidence.CompositorTrace ||
            attempt.MappingUncertaintyMilliseconds != 0 ? MetricVerdict.Incomplete : pacing;
        var rate = coverage != MetricVerdict.Pass ? MetricVerdict.Incomplete :
            ratio >= .99m && ratio <= 1.01m ? MetricVerdict.Pass : MetricVerdict.Fail;
        return new(coverage, pacing, browser, rate, identities.Count, slots, missed, fps, ratio, p95, p99);
    }

    public static MetricVerdict WarmupBeforeFreshRun(
        IReadOnlyList<AttemptObservation> warmup, AttemptObservation measured)
    {
        ArgumentNullException.ThrowIfNull(warmup);
        ArgumentNullException.ThrowIfNull(measured);
        decimal active = 0;
        decimal previous = -1;
        var ids = new HashSet<AttemptId>();
        foreach (var attempt in warmup)
        {
            if (attempt is null || attempt.Id.Value == Guid.Empty || !ids.Add(attempt.Id) ||
                attempt.Id == measured.Id || attempt.Purpose != AttemptPurpose.Warmup ||
                !attempt.StartAcknowledged || !attempt.StopObserved || !attempt.Continuous ||
                attempt.End == AttemptEnd.Interrupted || !Enum.IsDefined(attempt.End) ||
                attempt.StartMilliseconds < previous || attempt.EndMilliseconds <= attempt.StartMilliseconds ||
                attempt.EndMilliseconds >= measured.StartMilliseconds || attempt.FirstTick != 0 ||
                attempt.LastTick <= 0 || attempt.LastTick > 3600 ||
                (attempt.End == AttemptEnd.Timeout && attempt.LastTick != 3600))
                return MetricVerdict.Fail;
            active += attempt.EndMilliseconds - attempt.StartMilliseconds;
            previous = attempt.EndMilliseconds;
        }
        return active >= 10000 && measured.Purpose == AttemptPurpose.Measurement &&
            measured.Id.Value != Guid.Empty && measured.FirstTick == 0 ?
                MetricVerdict.Pass : MetricVerdict.Incomplete;
    }

    public static ThermalMetrics AnalyzeThermal(ThermalContract contract,
        IReadOnlyList<TimelineInterval> timeline)
    {
        ArgumentNullException.ThrowIfNull(contract);
        ArgumentNullException.ThrowIfNull(timeline);
        if (contract.MinimumDurationMilliseconds < 300000 ||
            contract.MinimumActiveFraction <= 0 || contract.MinimumActiveFraction > 1 ||
            contract.MaximumGapMilliseconds <= 0 || timeline.Count == 0 || timeline[0] is null)
            throw new InvalidDataException("Invalid thermal contract or empty timeline.");
        decimal active = 0, maxGap = 0, gap = 0;
        var previous = timeline[0].StartMilliseconds;
        var start = previous;
        var continuity = MetricVerdict.Pass;
        var ids = new HashSet<AttemptId>();
        foreach (var interval in timeline)
        {
            if (interval is null || !Enum.IsDefined(interval.Phase) ||
                interval.StartMilliseconds < 0 || interval.EndMilliseconds <= interval.StartMilliseconds ||
                interval.StartMilliseconds < previous)
                throw new InvalidDataException("Invalid thermal interval.");
            if (interval.StartMilliseconds != previous) continuity = MetricVerdict.Incomplete;
            if (interval.Phase == TimelinePhase.Hidden) continuity = MetricVerdict.Fail;
            if (interval.Phase == TimelinePhase.Active)
            {
                if (interval.Attempt is not { } id || id.Value == Guid.Empty || !ids.Add(id))
                    throw new InvalidDataException("Missing or stitched thermal attempt identity.");
                active += interval.EndMilliseconds - interval.StartMilliseconds;
                gap = 0;
            }
            else
            {
                if (interval.Attempt is not null)
                    throw new InvalidDataException("Lifecycle intervals cannot claim an active attempt.");
                gap += interval.EndMilliseconds - interval.StartMilliseconds;
                maxGap = Math.Max(maxGap, gap);
            }
            previous = interval.EndMilliseconds;
        }
        var duration = previous - start;
        var fraction = active / duration;
        var engagement = duration < contract.MinimumDurationMilliseconds ? MetricVerdict.Incomplete :
            fraction < contract.MinimumActiveFraction || maxGap > contract.MaximumGapMilliseconds ?
                MetricVerdict.Fail : MetricVerdict.Pass;
        return new(continuity, engagement, duration, fraction, maxGap);
    }
}
